前端 V7 流式 Fetch 代码实现 (POST)

My_V7_Fetch_Decoder.js 

// 这是一个适配 V7 的异步 POST 流式传输处理模板
async function sendV7StreamRequest(messageContent, conversationId) {
    const url = '/api/chat/stream';

    try {
        const response = await fetch(url, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                message: messageContent,
                conversationId: conversationId || ""
            })
        });

        if (!response.ok) throw new Error(`HTTP 报错！状态码: ${response.status}`);

        const reader = response.body.getReader();
        const decoder = new TextDecoder("utf-8");
        let buffer = "";

        while (true) {
            const { value, done } = await reader.read();
            if (done) break;
            
            buffer += decoder.decode(value, { stream: true });
            let lineEnd;
            
            // 逐行切分文本块，安全解码
            while ((lineEnd = buffer.indexOf("\n")) !== -1) {
                const line = buffer.slice(0, lineEnd).trim();
                buffer = buffer.slice(lineEnd + 1);
                
                if (line.startsWith("data:")) {
                    const dataStr = line.substring(5).trim();
                    try {
                        const parsed = JSON.parse(dataStr);
                        
                        // 1. 判断 token 类别并分流处理
                        if (typeof parsed === 'string') {
                            console.log("追加普通 Token:", parsed);
                        } else if (parsed.type === 'conversation_id') {
                            console.log("捕获会话 ID:", parsed.conversationId);
                        } else if (parsed.type === 'done') {
                            console.log("会话传输结束，最终文本:", parsed.text);
                        } else if (parsed.type === 'error') {
                            console.error("服务端异常:", parsed.message);
                        }
                    } catch (e) {
                        // 兜底：解不出来非标准 JSON 时，原样输出
                        console.log("原始裸文本:", dataStr);
                    }
                }
            }
        }
    } catch (error) {
        console.error('联调报错:', error);
    }
}

Go 后端 V7 会话持久化与 SSE 协议模型

service/agent.go (Go V7 SSE 架构范例) 
package service

import (
    "encoding/json"
    "fmt"
    "net/http"
)

type ChatRequest struct {
    Message        string `json:"message"`
    ConversationID string `json:"conversationId"`
}

type SSEFrame struct {
    Type           string `json:"type,omitempty"`
    ConversationID string `json:"conversationId,omitempty"`
    Text           string `json:"text,omitempty"`
    Message        string `json:"message,omitempty"`
}

func HandleChatStreamV7(w http.ResponseWriter, r *http.Request) {
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.Header().Set("Access-Control-Allow-Origin", "*")

    var req ChatRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, "Bad JSON", http.StatusBadRequest)
        return
    }

    flusher, _ := w.(http.Flusher)

    // 1. 发送首帧：携带 UUID (会话持久化)
    sessionID := req.ConversationID
    if sessionID == "" {
        sessionID = "session-uuid-v7-demo-999"
    }
    frame1, _ := json.Marshal(SSEFrame{Type: "conversation_id", ConversationID: sessionID})
    fmt.Fprintf(w, "data: %s\n\n", string(frame1))
    flusher.Flush()

    // 2. 发送逐字 Token (纯 JSON 字符串格式)
    tokens := []string{"Go", " ", "v7", " ", "会话", " ", "持久", " ", "解码", "!"}
    for _, t := range tokens {
        tokenBytes, _ := json.Marshal(t) // 这会产生带有双引号的 JSON 字符串，例如 "\"Go\""
        fmt.Fprintf(w, "data: %s\n\n", string(tokenBytes))
        flusher.Flush()
    }

    // 3. 终结帧发送
    frameDone, _ := json.Marshal(SSEFrame{Type: "done", Text: "Go v7 会话持久解码!"})
    fmt.Fprintf(w, "data: %s\n\n", string(frameDone))
    flusher.Flush()
}

📚 HTTP 前后端联调知识宝库

1. 流式响应 (SSE) 的核心要义

大语言模型的打字机流式输出，并不需要复杂的 WebSocket 握手。通过传统的 HTTP 报文，并将响应头设置为 Content-Type: text/event-stream 即可。前端使用原生的 fetch() 配合 reader.read()，在 TCP 报文不断到达时持续提取并解析文本行，即可实现极低延时的流渲染。

2. GET 与 POST 的深层技术差异

  • GET (获取资源)：不包含 Request Body (请求体)。参数直接携带在 URL 查询字符串中 (如 ?user=go)。在 Go 端使用 r.URL.Query().Get("user") 来提取。
  • POST (新建/传输资源)：参数存存放于 Request Body (请求体) 中，使用标准的序列化（通常是 JSON 文本）。Go 端读取 r.Body 并使用 json.NewDecoder(r.Body).Decode(&payload) 反序列化。

3. CORS 同源策略拦截与原理

浏览器出于安全考量，当域名、协议、或者端口不一致时，即为跨域（CORS）。你的前端页面运行在 localhost:端口A（或者本地文件协议），而 Go 后端运行在 localhost:8080。

如何优雅通关：最科学的方法，是让 Go 的 handleIndex（即 GET /）来提供当前 HTML 网页资源，这样当前页面与接口彻底同源。

前端流式与普通 Fetch 代码实现

My_Request_Implement.js 

// 这是一个融合了流式(SSE)解析和普通JSON应答的安全请求模板，完全避免 Stream 已经被读取的崩溃 bug
async function sendRequestToGoServer(messageContent) {
    const url = '/api/chat';

    try {
        const response = await fetch(url, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ message: messageContent })
        });

        if (!response.ok) throw new Error(`HTTP 报错！状态码: ${response.status}`);

        const contentType = response.headers.get('content-type') || '';
        
        // 核心：如果是 Server-Sent Events 流
        if (contentType.includes('text/event-stream')) {
            const reader = response.body.getReader();
            const decoder = new TextDecoder("utf-8");
            let buffer = "";

            while (true) {
                const { value, done } = await reader.read();
                if (done) break;
                
                buffer += decoder.decode(value, { stream: true });
                let lineEnd;
                
                // 逐行解析 SSE data 标志
                while ((lineEnd = buffer.indexOf("\n")) !== -1) {
                    const line = buffer.slice(0, lineEnd).trim();
                    buffer = buffer.slice(lineEnd + 1);
                    
                    if (line.startsWith("data:")) {
                        const dataStr = line.substring(5).trim();
                        if (dataStr === "[DONE]") break;
                        
                        try {
                            const parsed = JSON.parse(dataStr);
                            console.log("收到流片段:", parsed.reply);
                        } catch(e) {
                            console.log("收到非JSON行:", dataStr);
                        }
                    }
                }
            }
        } else {
            // 普通响应：先用 text() 消费流缓冲，再安全地反序列化 JSON，防 already read
            const resText = await response.text();
            try {
                const data = JSON.parse(resText);
                console.log("正常 JSON 数据:", data);
            } catch(e) {
                console.log("正常纯文本数据:", resText);
            }
        }
    } catch (error) {
        console.error('通信错误:', error);
    }
}

对应 Go 后端 SSE 路由匹配与输出逻辑

stream.go (Go SSE 后端架构范例) 
package main

import (
    "encoding/json"
    "fmt"
    "net/http"
    "time"
)

func handleChatStream(w http.ResponseWriter, r *http.Request) {
    // 1. 声明协议为 Server-Sent Events (SSE) 流
    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.Header().Set("Access-Control-Allow-Origin", "*") // 放行跨域

    // 2. 检查底层是否支持 Flusher
    flusher, ok := w.(http.Flusher)
    if !ok {
        http.Error(w, "Streaming unsupported", http.StatusInternalServerError)
        return
    }

    words := []string{"Hello", " ", "from", " ", "Go", " ", "SSE", " ", "Flusher", "!"}
    
    // 3. 逐字向下游刷新（Flush）发送 SSE 格式报文
    for _, word := range words {
        payload := map[string]string{"reply": word}
        jsonBytes, _ := json.Marshal(payload)
        
        // SSE 严格以 "data: 内容\n\n" 作为一帧的标识
        fmt.Fprintf(w, "data: %s\n\n", string(jsonBytes))
        
        flusher.Flush() // 瞬间冲刷 TCP 缓冲区发送到客户端
        time.Sleep(150 * time.Millisecond)
    }

    // 4. 流终结标识
    fmt.Fprintf(w, "data: [DONE]\n\n")
    flusher.Flush()
}

📚 HTTP 前后端联调知识宝库

1. 流式响应 (SSE) 的核心要义

大语言模型的打字机流式输出，并不需要复杂的 WebSocket 握手。通过传统的 HTTP 报文，并将响应头设置为 Content-Type: text/event-stream 即可。前端使用原生的 fetch() 配合 reader.read()，在 TCP 报文不断到达时持续提取并解析文本行，即可实现极低延时的流渲染。

2. GET 与 POST 的深层技术差异

  • GET (获取资源)：不包含 Request Body (请求体)。参数直接携带在 URL 查询字符串中 (如 ?user=go)。在 Go 端使用 r.URL.Query().Get("user") 来提取。
  • POST (新建/传输资源)：参数存存放于 Request Body (请求体) 中，使用标准的序列化（通常是 JSON 文本）。Go 端读取 r.Body 并使用 json.NewDecoder(r.Body).Decode(&payload) 反序列化。

3. CORS 同源策略拦截与原理

浏览器出于安全考量，当域名、协议、或者端口不一致时，即为跨域（CORS）。你的前端页面运行在 localhost:端口A（或者本地文件协议），而 Go 后端运行在 localhost:8080。

如何优雅通关：最科学的方法，是让 Go 的 handleIndex（即 GET /）来提供当前 HTML 网页资源，这样当前页面与接口彻底同源。
